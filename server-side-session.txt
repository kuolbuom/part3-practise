const jwt = require("jsonwebtoken");
const { SECRET } = require("../util/config");
const User = require("../models/user");
const Session = require("../models/session");

const tokenExtractor = async (req, res, next) => {
  const authorization = req.get("authorization");

  if (!authorization || !authorization.toLowerCase().startsWith("bearer ")) {
    return res.status(401).json({ error: "token missing" });
  }

  const token = authorization.substring(7);

  let decodedToken;

  try {
    decodedToken = jwt.verify(token, SECRET);
  } catch {
    return res.status(401).json({ error: "token invalid or expired" });
  }

  // Check whether the token exists in the sessions table
  const session = await Session.findOne({
    where: { token },
  });

  if (!session) {
    return res.status(401).json({ error: "session expired or invalid" });
  }

  // Retrieve the user, including disabled users
  const user = await User.unscoped().findByPk(session.userId);

  if (!user || user.disabled || user.id !== decodedToken.id) {
    return res.status(401).json({ error: "user disabled or session invalid" });
  }

  req.decodedToken = decodedToken;

  next();
};

const isAdmin = async (req, res, next) => {
  const user = await User.findByPk(req.decodedToken.id);

  if (!user || !user.admin) {
    return res.status(401).json({ error: "operation not allowed" });
  }

  next();
};

module.exports = { tokenExtractor, isAdmin };

Let's break down your JWT token. First, one small correction: signature, not signiture.

A JWT (JSON Web Token) has three parts separated by two dots (.):
Your JWT structure
1. Header

eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9

2. Payload

eyJ1c2VybmFtZSI6InNlY29uZHRlc3R1c2VyQGdtYWlsLmNvbSIsImlkIjoyLCJpYXQiOjE3OTM5NDgxNDB9

3. Signature

j_O8WggEmcP7eoaOOCc1mgChG9gShlqd4f1uRDYMJ2E


1. Header — what algorithm is used?
When decoded, your header becomes:
{
  "alg": "HS256",
  "typ": "JWT"
}


- alg: The algorithm used to sign the token.
- HS256: HMAC using SHA-256.
- typ: The token type, JWT.
Think of this as identifying the security method used to create the token.
2. Payload — what information is inside?
Your payload decodes to:
{
  "username": "secondtestuser@gmail.com",
  "id": 2,
  "iat": 1793948140
}


Let's understand each property:
Property	             Meaning
username	             The user's username
id	                   The user's database ID
iat	                   Issued At — when the token was created, as a Unix timestamp

Important: The payload is encoded, not encrypted. Anyone who has the token can decode and read it. Never put passwords or secrets inside it.
3. Signature — the most important part
This is the signature:
j_O8WggEmcP7eoaOOCc1mgChG9gShlqd4f1uRDYMJ2E


Your backend creates this signature using three things:
1. The encoded header.
2. The encoded payload.
3. Your secret key (SECRET).
Conceptually, it works like this:
signature = HMAC_SHA256(
  encodedHeader + "." + encodedPayload,
  SECRET
);


The signature acts like a tamper detector.

Imagine someone changes the payload:
{
  "username": "secondtestuser@gmail.com",
  "id": 2,
  "admin": true
}


They cannot simply change the payload and keep the original signature. When your backend verifies the token, the signature will not match.
4. How your backend verifies it
In your project, you have:
decodedToken = jwt.verify(request.token, SECRET);


The jsonwebtoken library:
1. Reads the header and payload.
2. Uses your SECRET to calculate the expected signature.
3. Compares it with the signature attached to the token.
4. Rejects the token if verification fails.
If verification succeeds, it returns the decoded payload.
5. How this connects to your session table
Remember, your project has another security check:
const session = await Session.findOne({
  where: { token: request.token },
});

if (!session) {
  return response.status(401).json({
    error: "session expired or invalid",
  });
}


This gives you two separate checks:
Request arrives with JWT
Check 1: Signature

Is the JWT authentic and valid?


Check 2: Database session

Does this token still exist in the sessions table?


Allow request if both checks pass

And now you have a third check:

Is the user disabled?
if (user.disabled) {
  return response.status(403).json({
    error: "account disabled",
  });
}


So remember:
- Signature: Is this token authentic?
- Session: Is this login session still active?
- Disabled: Is this user allowed to access the application?
